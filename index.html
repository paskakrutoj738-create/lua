-- === ДОПОЛНИТЕЛЬНЫЕ НАСТРОЙКИ MM2 === Settings.KillAll = false Settings.PreRoles = false local roleObjects = {} local killConnection = nil local roleConnection = nil local ROLE_COLORS = { Murderer = Color3.fromRGB(255, 60, 80), Sheriff = Color3.fromRGB(60, 150, 255), Hero = Color3.fromRGB(255, 210, 60), Innocent = Color3.fromRGB(70, 220, 130) } -- === ПОЛУЧЕНИЕ РОЛЕЙ ИЗ ДАННЫХ ИГРЫ === local function getRoleData() -- В разных версиях MM2 структура данных может отличаться. local candidates = {} local replicatedStorage = game:GetService("ReplicatedStorage") local remotes = replicatedStorage:FindFirstChild("Remotes") if remotes then local gameplay = remotes:FindFirstChild("Gameplay") if gameplay then local data = gameplay:FindFirstChild("PlayerData") if data then table.insert(candidates, data) end end end local direct = replicatedStorage:FindFirstChild("PlayerData") if direct then table.insert(candidates, direct) end for _, container in ipairs(candidates) do local result = {} for _, player in ipairs(Players:GetPlayers()) do local item = container:FindFirstChild(player.Name) if item then local roleValue = item:FindFirstChild("Role") if roleValue and typeof(roleValue.Value) == "string" then result[player] = roleValue.Value end end end if next(result) then return result end end return nil end -- === РОЛИ ДО ВЫДАЧИ ОРУЖИЯ === local function clearRoleESP() for _, object in pairs(roleObjects) do if object then object:Destroy() end end roleObjects = {} end local function updatePreRoles() if not Settings.PreRoles then clearRoleESP() return end local roles = getRoleData() if not roles then return end for player, role in pairs(roles) do if player ~= LocalPlayer and player.Character then local head = player.Character:FindFirstChild("Head") if head then local label = roleObjects[player] if not label or not label.Parent then label = Instance.new("BillboardGui") label.Name = "LUAVERSE_Role" label.Size = UDim2.new(0, 150, 0, 35) label.StudsOffset = Vector3.new(0, 3, 0) label.AlwaysOnTop = true label.Adornee = head label.Parent = head local text = Instance.new("TextLabel") text.Name = "RoleText" text.Size = UDim2.new(1, 0, 1, 0) text.BackgroundTransparency = 1 text.Font = Enum.Font.GothamBold text.TextSize = 15 text.TextStrokeTransparency = 0 text.TextStrokeColor3 = Color3.new(0, 0, 0) text.Parent = label roleObjects[player] = label end local text = label:FindFirstChild("RoleText") if text then text.Text = role text.TextColor3 = ROLE_COLORS[role] or Color3.fromRGB(255, 255, 255) end end end end end -- === ПРОВЕРКА: ТЫ УБИЙЦА? === local function isLocalMurderer() local character = LocalPlayer.Character local backpack = LocalPlayer:FindFirstChildOfClass("Backpack") local containers = {character, backpack} for _, container in ipairs(containers) do if container then for _, tool in ipairs(container:GetChildren()) do if tool:IsA("Tool") and tool.Name == "Knife" then return true end end end end local roles = getRoleData() if roles and roles[LocalPlayer] == "Murderer" then return true end return false end -- === АВТОУБИЙСТВО ВСЕХ === local function getKnife() local character = LocalPlayer.Character if character then local knife = character:FindFirstChild("Knife") if knife and knife:IsA("Tool") then return knife end end local backpack = LocalPlayer:FindFirstChildOfClass("Backpack") if backpack then local knife = backpack:FindFirstChild("Knife") if knife and knife:IsA("Tool") then return knife end end return nil end local function startKillAll() if killConnection then return end killConnection = task.spawn(function() while Settings.KillAll and ScreenGui.Parent do local character = LocalPlayer.Character local humanoid = character and character:FindFirstChildOfClass("Humanoid") local root = character and character:FindFirstChild("HumanoidRootPart") local knife = getKnife() if humanoid and root and knife and isLocalMurderer() then -- Берём нож в руки if knife.Parent ~= character then humanoid:EquipTool(knife) task.wait(0.15) end for _, target in ipairs(Players:GetPlayers()) do if not Settings.KillAll then break end if target ~= LocalPlayer and target.Character then local targetRoot = target.Character:FindFirstChild("HumanoidRootPart") local targetHumanoid = target.Character:FindFirstChildOfClass("Humanoid") if targetRoot and targetHumanoid and targetHumanoid.Health > 0 then -- Подлетаем к цели root.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 2) task.wait(0.12) -- Активируем нож pcall(function() knife:Activate() end) -- Пытаемся нанести удар через доступный Remote local stab = knife:FindFirstChild("Stab") if stab and stab:IsA("RemoteEvent") then pcall(function() stab:FireServer("Down") end) end task.wait(0.25) end end end end task.wait(0.4) end killConnection = nil end) end local function toggleKillAll(state) Settings.KillAll = state if state then startKillAll() end end local function togglePreRoles(state) Settings.PreRoles = state if state then updatePreRoles() else clearRoleESP() end end -- Обновление ролей task.spawn(function() while ScreenGui.Parent do updatePreRoles() task.wait(0.5) end end) -- === КНОПКИ В МЕНЮ === -- Если старые кнопки уже занимают позиции 50–200, -- увеличь высоту MainFrame до 420. MainFrame.Size = UDim2.new(0, 280, 0, 420) createToggle("💀 Kill All", 250, toggleKillAll) createToggle("🎭 Roles ESP", 300, togglePreRoles) -- Очистка при закрытии меню closeBtn.MouseButton1Click:Connect(function() Settings.KillAll = false Settings.PreRoles = false clearRoleESP() if killConnection then task.cancel(killConnection) killConnection = nil end end)
